// Pattern: CHANGE_ME
// Difficulty: Easy
// Problem: 3731. Find Missing Elements
// Link: https://leetcode.com/problems/find-missing-elements

class Solution:
    def findMissingElements(self, nums: List[int]) -> List[int]:
        nums.sort()
        ans = []
        for x, y in pairwise(nums):
            ans.extend(range(x + 1, y))
        return ans
