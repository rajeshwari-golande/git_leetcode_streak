// Problem: 3870. Count Commas in Range
// Link: https://leetcode.com/problems/count-commas-in-range

class Solution:
    def countCommas(self, n: int) -> int:
        res = 0
        for a in range(1, n + 1):
            if a > 999:
                res += 1
        return res
