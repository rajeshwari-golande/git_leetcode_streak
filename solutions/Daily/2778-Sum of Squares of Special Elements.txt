// Problem: 2778. Sum of Squares of Special Elements
// Link: https://leetcode.com/problems/sum-of-squares-of-special-elements

class Solution:
    def sumOfSquares(self, nums: List[int]) -> int:
        n = len(nums)
        ans = 0
        for i in range(n):
            if n % (i + 1) == 0:
                ans += nums[i] ** 2
        return ans
