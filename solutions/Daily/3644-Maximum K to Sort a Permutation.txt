// Pattern: CHANGE_ME
// Difficulty: Medium
// Problem: 3644. Maximum K to Sort a Permutation
// Link: https://leetcode.com/problems/maximum-k-to-sort-a-permutation

class Solution:
    def sortPermutation(self, nums: List[int]) -> int:
        n = len(nums)
        res = -1
        for i in range(n):
            if nums[i] == i:
                continue
            res &= nums[i]
        return 0 if res == -1 else res
