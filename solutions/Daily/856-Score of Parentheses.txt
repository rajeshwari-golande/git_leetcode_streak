// Pattern: CHANGE_ME
// Difficulty: Medium
// Problem: 856. Score of Parentheses
// Link: https://leetcode.com/problems/score-of-parentheses

class Solution:
    def scoreOfParentheses(self, s: str) -> int:
        stack = [0]

        for ch in s:
            if ch == '(':
                stack.append(0)
            else:
                inside = stack.pop()

                if inside == 0:
                    stack.append(stack.pop() + 1)
                else:
                    stack.append(stack.pop() + 2 * inside)

        return stack.pop()
