// Problem: 3536. Maximum Product of Two Digits
// Link: https://leetcode.com/problems/maximum-product-of-two-digits

class Solution:
    def maxProduct(self, n: int) -> int:
        first, second = 0, 0
        while n > 0:
            x = n % 10
            if x > first:
                first, second = x, first
            elif x > second:
                second = x
            n //= 10
        return first * second
