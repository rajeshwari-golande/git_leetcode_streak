# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    #two BST iterator solutin pending


    #iterative BFS :
    def findTarget(self, root: Optional[TreeNode], k: int) -> bool:
        if not root:
            return False
        seen=set()
        q=deque([root])
        while q:
            node=q.popleft()
            complement=k-node.val
            if complement in seen:
                return True
            seen.add(node.val)
            if node.left:
                q.append(node.left)
            if node.right:
                q.append(node.right)
        return False
